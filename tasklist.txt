### Task: Add "Copy Attachment" to Context Menu in `ResponsePane`

#### Objective
In the `AI Response` pane (`ResponsePane`), when a user right-clicks on an attachment area (such as the attached file code/text box or an image), the context menu should display an action: **"Copy Attachment"**.

Clicking "Copy Attachment" should copy the full content of the attachment to the system clipboard:
- If the attachment is **text or code** (e.g., `.txt`, `.cpp`, `.py`, `.json`, etc.), copy the complete raw text of the file (`QString::fromUtf8(data)`).
- If the attachment is an **image** (or an inline image), copy the image to the clipboard as a `QPixmap` / `QImage`.
- Normal text editing context menu actions ("Copy" for selected text, "Select All") must continue to work normally.

#include <QClipboard>
#include <QGuiApplication>
#include <QMenu>

bool ResponsePane::isPosOverAttachment(const QPoint &pos, QString &outFileId, QString &outImageName) const
{
    outFileId.clear();
    outImageName.clear();

    QTextCursor cursor = m_browser->cursorForPosition(pos);
    int p = cursor.position();

    // 1. Check if over Drive Document attachment (code/text/image)
    if (m_attachmentStartPos != -1 && m_attachmentEndPos != -1)
    {
        if (p >= m_attachmentStartPos && p <= m_attachmentEndPos)
        {
            outFileId = m_currentPair.prompt.driveDocumentId;
            return true;
        }
    }

    // 2. Check if hovering over an image format (attached image or inline image)
    QTextCharFormat fmt = cursor.charFormat();
    if (fmt.isImageFormat())
    {
        outImageName = fmt.toImageFormat().name();
        return true;
    }

    return false;
}

void ResponsePane::onCustomContextMenuRequested(const QPoint &pos)
{
    QMenu *menu = m_browser->createStandardContextMenu(pos);

    QString fileId;
    QString imageName;
    if (isPosOverAttachment(pos, fileId, imageName))
    {
        QAction *copyAttAction = new QAction(QIcon::fromTheme("edit-copy"), tr("Copy Attachment"), menu);
        connect(copyAttAction, &QAction::triggered, this, [this, fileId, imageName]() {
            copyAttachmentToClipboard(fileId, imageName);
        });

        // Insert at the top of the menu before standard actions
        if (!menu->actions().isEmpty())
        {
            menu->insertAction(menu->actions().first(), copyAttAction);
            menu->insertSeparator(menu->actions().at(1));
        }
        else
        {
            menu->addAction(copyAttAction);
        }
    }

    menu->exec(m_browser->mapToGlobal(pos));
    delete menu;
}

void ResponsePane::copyAttachmentToClipboard(const QString &fileId, const QString &imageResourceName)
{
    QClipboard *clipboard = QGuiApplication::clipboard();
    if (!clipboard) return;

    // Case 1: Drive attachment loaded in memory cache
    if (!fileId.isEmpty() && m_attachmentCache.contains(fileId))
    {
        const auto &att = m_attachmentCache[fileId];
        const QByteArray &data = att.first;
        const QString &mime = att.second;

        if (mime.startsWith("image/"))
        {
            QPixmap pix;
            if (pix.loadFromData(data))
                clipboard->setPixmap(pix);
        }
        else
        {
            // Text/code file (e.g. codetree.txt, README.md, source code)
            clipboard->setText(QString::fromUtf8(data));
        }
        return;
    }

    // Case 2: Image resource embedded in document (inline image)
    if (!imageResourceName.isEmpty())
    {
        QVariant res = m_browser->document()->resource(QTextDocument::ImageResource, QUrl(imageResourceName));
        if (res.isValid())
        {
            QImage img = res.value<QImage>();
            if (!img.isNull())
            {
                clipboard->setImage(img);
                return;
            }
            QPixmap pix = res.value<QPixmap>();
            if (!pix.isNull())
            {
                clipboard->setPixmap(pix);
                return;
            }
        }
    }
