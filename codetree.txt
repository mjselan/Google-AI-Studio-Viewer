#include "ConversationModel.h"
#include "app/Logger.h"
#include <QJsonArray>
#include <QJsonDocument>
#include <algorithm>

ConversationModel::ConversationModel(QObject *parent)
    : QObject(parent) {}

Conversation ConversationModel::parseConversation(const QString &fileId,
                                                   const QString &fileName,
                                                   const QJsonObject &root) const
{
    Conversation conv;
    conv.fileId = fileId;
    conv.fileName = fileName;

    // Run settings
    QJsonObject settings = root["runSettings"].toObject();
    conv.model = settings["model"].toString();
    conv.temperature = settings["temperature"].toDouble(1.0);
    conv.topP = settings["topP"].toDouble(0.95);
    conv.topK = settings["topK"].toInt(64);
    conv.maxOutputTokens = settings["maxOutputTokens"].toInt(8192);
    conv.systemInstruction = root["systemInstruction"].toObject();

    // Parse chunks
    QJsonArray chunks = root["chunkedPrompt"].toObject()["chunks"].toArray();
    for (const auto &chunk : chunks)
        conv.messages.append(Message::fromJson(chunk.toObject()));

    // Sort messages chronologically; user before model for same timestamp
    std::sort(conv.messages.begin(), conv.messages.end(),
        [](const Message &a, const Message &b)
        {
            if (a.createTime != b.createTime)
                return a.createTime < b.createTime;
            if (a.role != b.role)
                return a.role == "user";
            return false;
        });

    buildPairs(conv);

    // Log message order for attachment debugging
    for (int i = 0; i < conv.messages.size(); ++i)
    {
        const auto &m = conv.messages[i];
        if (!m.driveDocumentId.isEmpty() || m.role == "user")
            Logger::instance().info(
                QString("[Attachment] Msg[%1] role='%2' doc='%3' text='%4' time=%5")
                    .arg(i)
                    .arg(m.role)
                    .arg(m.driveDocumentId)
                    .arg(m.text.left(40))
                    .arg(m.createTime.toString(Qt::ISODate)));
    }

    return conv;
}

void ConversationModel::buildPairs(Conversation &conv) const
{
    Message currentPrompt;
    bool hasPrompt = false;

    for (const auto &msg : conv.messages)
    {
        if (msg.role == "user")
        {
            if (hasPrompt && msg.createTime == currentPrompt.createTime)
            {
                // Same timestamp — merge: combine driveDocumentId and text
                bool currentHasDoc = !currentPrompt.driveDocumentId.isEmpty();
                bool newHasDoc = !msg.driveDocumentId.isEmpty();
                bool currentHasText = !currentPrompt.text.isEmpty();
                bool newHasText = !msg.text.isEmpty();

                Logger::instance().info(
                    QString("[Attachment] Merge check at %1: "
                            "current(doc='%2' text=%3) new(doc='%4' text=%5)")
                        .arg(msg.createTime.toString(Qt::ISODate))
                        .arg(currentPrompt.driveDocumentId)
                        .arg(currentHasText)
                        .arg(msg.driveDocumentId)
                        .arg(newHasText));

                // Prefer the message that has text as the base
                if (newHasText && !currentHasText)
                {
                    QString docId = currentPrompt.driveDocumentId;
                    QString imgData = currentPrompt.inlineImageData;
                    QString imgMime = currentPrompt.inlineImageMime;
                    currentPrompt = msg;
                    if (currentPrompt.driveDocumentId.isEmpty())
                        currentPrompt.driveDocumentId = docId;
                    if (currentPrompt.inlineImageData.isEmpty())
                    {
                        currentPrompt.inlineImageData = imgData;
                        currentPrompt.inlineImageMime = imgMime;
                    }
                    Logger::instance().info(
                        QString("[Attachment] Merged doc-only into text prompt: docId='%1' text='%2'")
                            .arg(currentPrompt.driveDocumentId)
                            .arg(currentPrompt.text.left(60)));
                }
                else if (currentHasText && !newHasText)
                {
                    if (currentPrompt.driveDocumentId.isEmpty())
                        currentPrompt.driveDocumentId = msg.driveDocumentId;
                    if (currentPrompt.inlineImageData.isEmpty())
                    {
                        currentPrompt.inlineImageData = msg.inlineImageData;
                        currentPrompt.inlineImageMime = msg.inlineImageMime;
                    }
                    Logger::instance().info(
                        QString("[Attachment] Merged doc into text prompt: docId='%1' text='%2'")
                            .arg(currentPrompt.driveDocumentId)
                            .arg(currentPrompt.text.left(60)));
                }
                else if (!currentHasText && !newHasText)
                {
                    // Both doc-only — keep whichever has the driveDocumentId
                    if (currentPrompt.driveDocumentId.isEmpty())
                        currentPrompt.driveDocumentId = msg.driveDocumentId;
                }
                // else: both have text — keep current (first one wins)
            }
            else if (hasPrompt)
            {
                currentPrompt = msg;
            }
            else
            {
                currentPrompt = msg;
            }
            hasPrompt = true;
        }
        else if (msg.role == "model" && hasPrompt)
        {
            if (conv.pairs.isEmpty() ||
                conv.pairs.last().prompt.createTime != currentPrompt.createTime)
            {
                Conversation::PromptResponsePair pair;
                pair.prompt = currentPrompt;
                pair.responses.append(msg);
                conv.pairs.append(pair);
                if (!pair.prompt.driveDocumentId.isEmpty())
                    Logger::instance().info(
                        QString("[Attachment] Pair %1 created with docId='%2' text='%3'")
                            .arg(conv.pairs.size() - 1)
                            .arg(pair.prompt.driveDocumentId)
                            .arg(pair.prompt.text.left(60)));
            }
            else
            {
                conv.pairs.last().responses.append(msg);
            }
        }
    }

    // Flush last prompt if no model response followed
    if (hasPrompt)
    {
        bool found = false;
        for (const auto &p : conv.pairs)
        {
            if (p.prompt.createTime == currentPrompt.createTime)
            {
                found = true;
                break;
            }
        }
        if (!found)
        {
            Conversation::PromptResponsePair pair;
            pair.prompt = currentPrompt;
            conv.pairs.append(pair);
            if (!pair.prompt.driveDocumentId.isEmpty())
                Logger::instance().info(
                    QString("[Attachment] Final pair %1 (no response) with docId='%2'")
                        .arg(conv.pairs.size() - 1)
                        .arg(pair.prompt.driveDocumentId));
        }
    }
}

#include "CopilotApi.h"
#include <QJsonDocument>
#include <QJsonArray>
#include <QJsonObject>
#include <QNetworkRequest>
#include <QNetworkReply>
#include <QUrl>

CopilotApi::CopilotApi(QObject *parent)
    : QObject(parent)
    , m_net(new QNetworkAccessManager(this))
    , m_baseUrl("http://localhost:8000")
{
}

void CopilotApi::setBaseUrl(const QString &url)
{
    m_baseUrl = url;
}

void CopilotApi::fetchModels()
{
    QUrl url(m_baseUrl + "/v1/models");
    QNetworkRequest req(url);

    auto *reply = m_net->get(req);
    connect(reply, &QNetworkReply::finished, this, [this, reply]()
    {
        reply->deleteLater();

        auto doc = QJsonDocument::fromJson(reply->readAll());

        if (reply->error() != QNetworkReply::NoError)
        {
            QString errMsg = reply->errorString();
            emit connectionFailed(errMsg);
            return;
        }
        auto data = doc.object()["data"].toArray();
        if (!data.isEmpty())
        {
            m_modelName = data.first().toObject()["id"].toString();
            emit connected(m_modelName);
        }
        else
        {
            emit connectionFailed("No models returned");
        }
    });
}

void CopilotApi::sendMessage(const QString &prompt, const QString &conversationId)
{
    QJsonObject body;
    body["model"] = m_modelName.isEmpty() ? "copilot" : m_modelName;
    body["stream"] = false;

    QJsonArray messages;
    QJsonObject msg;
    msg["role"] = "user";
    msg["content"] = prompt;
    messages.append(msg);
    body["messages"] = messages;

    if (!conversationId.isEmpty())
        body["conversation_id"] = conversationId;

    QUrl url(m_baseUrl + "/v1/chat/completions");
    QNetworkRequest req(url);
    req.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");

    auto *reply = m_net->post(req, QJsonDocument(body).toJson());
    connect(reply, &QNetworkReply::finished, this, [this, reply]()
    {
        reply->deleteLater();

        auto doc = QJsonDocument::fromJson(reply->readAll());
        auto root = doc.object();

        if (reply->error() != QNetworkReply::NoError)
        {
            QString errMsg;
            if (root.contains("error"))
                errMsg = root["error"].toObject()["message"].toString();
            else
                errMsg = reply->errorString();

            emit errorOccurred(errMsg);
            return;
        }

        if (root.contains("error"))
        {
            QString errMsg = root["error"].toObject()["message"].toString();
            emit errorOccurred(errMsg);
            return;
        }

        QString content = root["choices"].toArray()
            .first().toObject()["message"].toObject()["content"].toString();
        QString convId = root["conversation_id"].toString();

        emit responseReceived(content, convId);
    });
}

#include <QUrlQuery>
#include <QDebug>

DriveApi::DriveApi(QObject *parent)
    : QObject(parent)
    , m_net(new QNetworkAccessManager(this))
{
}

void DriveApi::setAccessToken(const QString &token)
{
    m_token = token;
}

void DriveApi::sendGet(const QString &urlStr, std::function<void(const QJsonObject &)> callback)
{
    QUrl qurl(urlStr);
    QNetworkRequest req(qurl);
    req.setRawHeader("Authorization", ("Bearer " + m_token).toUtf8());
    req.setAttribute(QNetworkRequest::Http2AllowedAttribute, false);

    auto *reply = m_net->get(req);
    handleReply(reply, callback);
}

void DriveApi::handleReply(QNetworkReply *reply, std::function<void(const QJsonObject &)> callback)
{
    connect(reply, &QNetworkReply::finished, this, [this, reply, callback]()
    {
        reply->deleteLater();

        int status = reply->attribute(QNetworkRequest::HttpStatusCodeAttribute).toInt();

        if (status == 401)
        {
            PendingRequest pending;
            pending.url = reply->url().toString();
            pending.callback = callback;
            m_pendingRequests.append(pending);
            emit tokenNeedsRefresh();
            return;
        }

        if (reply->error() != QNetworkReply::NoError)
        {
            qWarning() << "Drive API error:" << reply->errorString();
            return;
        }

        auto doc = QJsonDocument::fromJson(reply->readAll());
        callback(doc.object());
    });
}

void DriveApi::retryPendingRequests()
{
    int count = m_pendingRequests.size();
    QList<PendingRequest> retries = m_pendingRequests;
    m_pendingRequests.clear();
    for (const auto &pending : retries)
        sendGet(pending.url, pending.callback);
}

void DriveApi::findFolder(const QString &folderName)
{
    QUrlQuery query;
    query.addQueryItem("q", QString("name='%1' and mimeType='application/vnd.google-apps.folder' and trashed=false").arg(folderName));
    query.addQueryItem("fields", "files(id,name)");
    query.addQueryItem("pageSize", "10");

    QUrl url("https://www.googleapis.com/drive/v3/files");
    url.setQuery(query);

    sendGet(url.toString(), [this](const QJsonObject &response)
    {
        auto files = response["files"].toArray();
        if (files.isEmpty())
            emit folderNotFound();
        else
            emit folderFound(files.first().toObject()["id"].toString());
    });
}

void DriveApi::listConversations(const QString &folderId)
{
    QUrlQuery query;
    // Revert the q parameter to fetch all active files in the folder
    query.addQueryItem("q", QString("'%1' in parents and trashed=false").arg(folderId));
    // ADDED: Request mimeType and fileExtension from Google Drive
    query.addQueryItem("fields", "files(id,name,modifiedTime,mimeType,fileExtension)");
    query.addQueryItem("orderBy", "modifiedTime desc");
    query.addQueryItem("pageSize", "1000");

    QUrl url("https://www.googleapis.com/drive/v3/files");
    url.setQuery(query);

    sendGet(url.toString(), [this](const QJsonObject &response)
            {
                QList<DriveFile> files;
                for (const auto &f : response["files"].toArray())
                {
                    auto obj = f.toObject();
                    QString id = obj["id"].toString();
                    QString name = obj["name"].toString();
                    QString mime = obj["mimeType"].toString();
                    QString ext = obj["fileExtension"].toString().toLower();

                    // 1. Filter out obvious media/document MIME types
                    if (mime.startsWith("image/") ||
                        mime.startsWith("video/") ||
                        mime.startsWith("audio/") ||
                        mime == "application/pdf" ||
                        mime == "text/csv")
                    {
                        continue;
                    }

                    // 2. Filter out files with common attachment extensions.
                    // Google AI Studio conversations are saved without a file extension.
                    // If it has an extension like .json or .txt, it's a user-uploaded attachment.
                    if (!ext.isEmpty())
                    {
                        QStringList attachmentExts = {
                            "json", "txt", "pdf", "csv", "md", "png", "jpg", "jpeg",
                            "mp3", "mp4", "wav", "py", "cpp", "h", "js", "html"
                        };
                        if (attachmentExts.contains(ext)) {
                            continue;
                        }
                    }

                    DriveFile df;
                    df.fileId = id;
                    df.fileName = name;
                    df.modifiedTime = QDateTime::fromString(obj["modifiedTime"].toString(), Qt::ISODate);
                    files.append(df);
                }
                emit conversationListReady(files);
            });
}

void DriveApi::downloadConversation(const QString &fileId, const QString &fileName)
{
    QUrl url(QString("https://www.googleapis.com/drive/v3/files/%1?alt=media").arg(fileId));
    QNetworkRequest req(url);
    req.setRawHeader("Authorization", ("Bearer " + m_token).toUtf8());
    req.setAttribute(QNetworkRequest::Http2AllowedAttribute, false);

    auto *reply = m_net->get(req);
    connect(reply, &QNetworkReply::finished, this, [this, reply, fileId, fileName]()
    {
        reply->deleteLater();

        int status = reply->attribute(QNetworkRequest::HttpStatusCodeAttribute).toInt();
        if (status == 401)
        {
            PendingRequest pending;
            pending.url = reply->url().toString();
            pending.callback = [this, fileId, fileName](const QJsonObject &data)
            {
                emit conversationDownloaded(fileId, fileName, data);
            };
            m_pendingRequests.append(pending);
            emit tokenNeedsRefresh();
            return;
        }

        if (reply->error() != QNetworkReply::NoError)
        {
            emit apiError("Download failed: " + reply->errorString());
            return;
        }
        auto doc = QJsonDocument::fromJson(reply->readAll());
        emit conversationDownloaded(fileId, fileName, doc.object());
    });
}

void DriveApi::downloadAttachment(const QString &fileId)
{
    Logger::instance().info(
        QString("[Attachment] downloadAttachment: fileId='%1'").arg(fileId));
    QUrl url(QString("https://www.googleapis.com/drive/v3/files/%1?alt=media").arg(fileId));
    QNetworkRequest req(url);
    req.setRawHeader("Authorization", ("Bearer " + m_token).toUtf8());
    req.setAttribute(QNetworkRequest::Http2AllowedAttribute, false);

    auto *reply = m_net->get(req);
    connect(reply, &QNetworkReply::finished, this, [this, reply, fileId]()
    {
        reply->deleteLater();

        int status = reply->attribute(QNetworkRequest::HttpStatusCodeAttribute).toInt();
        if (status == 401)
        {
            PendingRequest pending;
            pending.url = reply->url().toString();
            pending.callback = [this, fileId](const QJsonObject &)
            {
                downloadAttachment(fileId);
            };
            m_pendingRequests.append(pending);
            emit tokenNeedsRefresh();
            return;
        }

        if (reply->error() != QNetworkReply::NoError)
        {
            Logger::instance().error(
                QString("[Attachment] downloadAttachment failed: fileId='%1' error='%2'")
                    .arg(fileId, reply->errorString()));
            emit apiError("Attachment download failed: " + reply->errorString());
            return;
        }
        QString mime = reply->header(QNetworkRequest::ContentTypeHeader).toString();
        QByteArray data = reply->readAll();
        Logger::instance().info(
            QString("[Attachment] downloadAttachment OK: fileId='%1' mime='%2' size=%3")
                .arg(fileId, mime).arg(data.size()));
        emit attachmentDownloaded(fileId, data, mime);
    });
}

AuthManager::AuthManager(QObject *parent)
    : QObject(parent)
    , m_net(new QNetworkAccessManager(this))
{
}

bool AuthManager::isAuthenticated() const
{
    return !m_accessToken.isEmpty();
}

QString AuthManager::accessToken() const { return m_accessToken; }

int AuthManager::expiresIn() const { return m_expiresIn; }

void AuthManager::loadConfig(const QString &configPath)
{
    QFile f(configPath);
    if (!f.open(QIODevice::ReadOnly))
    {
        emit authFailed("Could not open config file: " + configPath);
        return;
    }
    auto doc = QJsonDocument::fromJson(f.readAll());
    f.close();

    // Support both root-level keys and the "installed" wrapper from Google Cloud download
    QJsonObject obj = doc.object();
    bool isGoogleFormat = obj.contains("installed");
    if (isGoogleFormat)
        obj = obj["installed"].toObject();

    m_clientId = obj["client_id"].toString();
    m_clientSecret = obj["client_secret"].toString();

    // For the Google download format, force OOB redirect (our auth flow)
    m_redirectUri = obj["redirect_uri"].toString();
    if (m_redirectUri.isEmpty())
        m_redirectUri = "urn:ietf:wg:oauth:2.0:oob";

    m_scopes << "https://www.googleapis.com/auth/drive.readonly";

    loadSavedTokens();
}

void AuthManager::setCredentials(const QString &clientId, const QString &clientSecret)
{
    m_clientId = clientId;
    m_clientSecret = clientSecret;
    m_redirectUri = "urn:ietf:wg:oauth:2.0:oob";
    m_scopes << "https://www.googleapis.com/auth/drive.readonly";

    loadSavedTokens();
}

void AuthManager::loadSavedTokens()
{
    auto *app = qobject_cast<Application *>(qApp);
    if (!app)
        return;

    QSettings &s = app->settings();
    m_refreshToken = s.value("auth/refresh_token").toString();
    m_accessToken = s.value("auth/access_token").toString();
    m_expiresIn = s.value("auth/expires_in").toInt();

    QString acquiredStr = s.value("auth/token_acquired_at").toString();
    if (!acquiredStr.isEmpty())
        m_tokenAcquiredAt = QDateTime::fromString(acquiredStr, Qt::ISODate);

    if (!m_accessToken.isEmpty() && m_tokenAcquiredAt.isValid())
    {
        qint64 elapsedSec = m_tokenAcquiredAt.secsTo(QDateTime::currentDateTimeUtc());
        int remainingSec = m_expiresIn - static_cast<int>(elapsedSec);
        if (remainingSec <= 60)
        {
            m_accessToken.clear();
        }
    }
}

void AuthManager::authenticate()
{
    if (m_clientId.isEmpty())
    {
        emit authFailed("Client ID not configured. Edit config/oauth_config.json");
        return;
    }

    // If we already have a valid access token, nothing to do
    if (!m_accessToken.isEmpty())
    {
        int remainingSec = m_expiresIn;
        if (m_tokenAcquiredAt.isValid())
        {
            qint64 elapsedSec = m_tokenAcquiredAt.secsTo(QDateTime::currentDateTimeUtc());
            remainingSec = m_expiresIn - static_cast<int>(elapsedSec);
        }
        int hours = remainingSec / 3600;
        int mins = (remainingSec % 3600) / 60;
        emit authSuccess(
            QString("Authenticated. Token expires in %1h %2m.").arg(hours).arg(mins));
        return;
    }

    // If we already have a refresh token stored from a previous authorization,
    // exchange it for a new access token rather than starting from scratch.
    if (!m_refreshToken.isEmpty())
    {
        refreshAccessToken();
        return;
    }

    QString url = buildAuthUrl();
    QDesktopServices::openUrl(QUrl(url));
    emit authCodeRequired(url);
}

void AuthManager::submitAuthCode(const QString &code)
{
    exchangeCodeForTokens(code);
}

void AuthManager::refreshAccessToken()
{
    if (m_refreshToken.isEmpty())
    {
        emit authFailed("No refresh token available");
        return;
    }

    QUrl url("https://oauth2.googleapis.com/token");
    QNetworkRequest req(url);
    req.setHeader(QNetworkRequest::ContentTypeHeader, "application/x-www-form-urlencoded");
    req.setAttribute(QNetworkRequest::Http2AllowedAttribute, false);

    QUrlQuery body;
    body.addQueryItem("client_id", m_clientId);
    body.addQueryItem("client_secret", m_clientSecret);
    body.addQueryItem("refresh_token", m_refreshToken);
    body.addQueryItem("grant_type", "refresh_token");

    auto *reply = m_net->post(req, body.toString(QUrl::FullyEncoded).toUtf8());
    connect(reply, &QNetworkReply::finished, this, [this, reply]()
            {
                reply->deleteLater();
                auto responseBody = reply->readAll();
                if (reply->error() != QNetworkReply::NoError)
                {
                    // If the refresh token is invalid or expired, clear it so the
                    // user can start from scratch next time.
                    signOut();
                    QString errMsg = "Token refresh failed: " + reply->errorString();
                    auto doc = QJsonDocument::fromJson(responseBody);
                    if (!doc.isNull() && doc.object().contains("error"))
                        errMsg += "\n" + doc.object()["error_description"].toString();
                    emit authFailed(errMsg);
                    return;
                }
                auto doc = QJsonDocument::fromJson(responseBody);
                auto obj = doc.object();
                QString access = obj["access_token"].toString();
                int expiresIn = obj["expires_in"].toInt();
                storeTokens(access, QString(), expiresIn);
                int hours = expiresIn / 3600;
                int mins = (expiresIn % 3600) / 60;
                emit tokenRefreshed();
                emit authSuccess(
                    QString("Token refreshed. Expires in %1h %2m.").arg(hours).arg(mins));
            });
}

void AuthManager::signOut()
{
    m_accessToken.clear();
    m_refreshToken.clear();
    m_expiresIn = 0;
    m_tokenAcquiredAt = QDateTime();
    auto *app = qobject_cast<Application *>(qApp);
    if (app)
    {
        QSettings &s = app->settings();
        s.remove("auth/refresh_token");
        s.remove("auth/access_token");
        s.remove("auth/expires_in");
        s.remove("auth/token_acquired_at");
        s.sync();
    }
}

void AuthManager::exchangeCodeForTokens(const QString &code)
{
    QUrl url("https://oauth2.googleapis.com/token");
    QNetworkRequest req(url);
    req.setHeader(QNetworkRequest::ContentTypeHeader, "application/x-www-form-urlencoded");
    req.setAttribute(QNetworkRequest::Http2AllowedAttribute, false);

    QUrlQuery body;
    body.addQueryItem("code", code);
    body.addQueryItem("client_id", m_clientId);
    body.addQueryItem("client_secret", m_clientSecret);
    body.addQueryItem("redirect_uri", m_redirectUri);
    body.addQueryItem("grant_type", "authorization_code");

    auto *reply = m_net->post(req, body.toString(QUrl::FullyEncoded).toUtf8());
    connect(reply, &QNetworkReply::finished, this, [this, reply]()
    {
        reply->deleteLater();
        auto responseBody = reply->readAll();
        if (reply->error() != QNetworkReply::NoError)
        {
            QString errMsg = "Token exchange failed: " + reply->errorString();
            auto doc = QJsonDocument::fromJson(responseBody);
            if (!doc.isNull() && doc.object().contains("error"))
                errMsg += "\n" + doc.object()["error_description"].toString();
            emit authFailed(errMsg);
            return;
        }
        auto doc = QJsonDocument::fromJson(responseBody);
        auto obj = doc.object();
        QString access = obj["access_token"].toString();
        QString refresh = obj["refresh_token"].toString();
        int expiresIn = obj["expires_in"].toInt();
        storeTokens(access, refresh, expiresIn);
        int hours = expiresIn / 3600;
        int mins = (expiresIn % 3600) / 60;
        emit authSuccess(
            QString("Authenticated. Token expires in %1h %2m.").arg(hours).arg(mins));
    });
}

void AuthManager::storeTokens(const QString &access, const QString &refresh, int expiresIn)
{
    m_accessToken = access;
    m_expiresIn = expiresIn;
    m_tokenAcquiredAt = QDateTime::currentDateTimeUtc();

    if (!refresh.isEmpty())
        m_refreshToken = refresh;

    // Always persist tokens
    auto *app = qobject_cast<Application *>(qApp);
    if (app)
    {
        QSettings &s = app->settings();
        s.setValue("auth/access_token", m_accessToken);
        s.setValue("auth/expires_in", m_expiresIn);
        s.setValue("auth/token_acquired_at", m_tokenAcquiredAt.toString(Qt::ISODate));
        if (!refresh.isEmpty())
            s.setValue("auth/refresh_token", m_refreshToken);
        s.sync();
    }
}

QString AuthManager::buildAuthUrl() const
{
    QUrlQuery query;
    query.addQueryItem("client_id", m_clientId);
    query.addQueryItem("redirect_uri", m_redirectUri);
    query.addQueryItem("response_type", "code");
    query.addQueryItem("scope", m_scopes.join(" "));
    query.addQueryItem("access_type", "offline");
    query.addQueryItem("prompt", "consent");

    QUrl url("https://accounts.google.com/o/oauth2/v2/auth");
    url.setQuery(query);
    return url.toString();
}

void Logger::init(const QString &filePath)
{

#ifdef NO_LOGGING
    return;
#endif
    QMutexLocker lock(&m_mutex);
    QDir().mkpath(QFileInfo(filePath).absolutePath());

    if (m_file.isOpen())
        m_file.close();

    m_file.setFileName(filePath);
    m_file.open(QIODevice::WriteOnly | QIODevice::Truncate | QIODevice::Text);
}

void Logger::log(Level level, const QString &message)
{
#ifdef NO_LOGGING
    return;
#endif
    QMutexLocker lock(&m_mutex);

    QString line = QString("[%1] [%2] %3\n")
        .arg(QDateTime::currentDateTime().toString("yyyy-MM-dd hh:mm:ss.zzz"))
        .arg(levelName(level))
        .arg(message);

    if (m_file.isOpen())
    {
        m_file.write(line.toUtf8());
        m_file.flush();
    }

    // Also print to stderr for immediate visibility
    std::cerr << line.toStdString();
}

QString Logger::levelName(Level level)
{
    switch (level)
    {
    case Debug:   return "DEBUG";
    case Info:    return "INFO";
    case Warning: return "WARN";
    case Error:   return "ERROR";
    }
    return "UNKNOWN";
}

Message Message::fromJson(const QJsonObject &obj)
{
    Message m;
    m.text = obj["text"].toString();
    m.role = obj["role"].toString();
    m.tokenCount = obj["tokenCount"].toInt();
    m.isThought = obj["isThought"].toBool(false);
    m.finishReason = obj["finishReason"].toString();
    m.parts = obj["parts"].toArray();

    // Parse createTime — handle Z suffix robustly
    QString ts = obj["createTime"].toString();
    m.createTime = QDateTime::fromString(ts, Qt::ISODateWithMs);
    if (!m.createTime.isValid() && ts.endsWith('Z'))
        m.createTime = QDateTime::fromString(ts.chopped(1) + "+00:00", Qt::ISODateWithMs);

    if (obj.contains("driveDocument"))
    {
        QJsonObject dd = obj["driveDocument"].toObject();
        m.driveDocumentId = dd["id"].toString();
    }
    else if (obj.contains("driveImage"))
    {
        QJsonObject di = obj["driveImage"].toObject();
        m.driveDocumentId = di["id"].toString();
    }

    // Check parts for inline image data (pasted images)
    for (int i = 0; i < m.parts.size(); ++i)
    {
        QJsonObject part = m.parts[i].toObject();
        QStringList keys = part.keys();

        if (part.contains("inlineData"))
        {
            QJsonObject inlineData = part["inlineData"].toObject();
            m.inlineImageMime = inlineData["mimeType"].toString();
            m.inlineImageData = inlineData["data"].toString();
        }
    }

    return m;
}

void ConversationListPane::refreshList()
{
    applySort();

    m_list->blockSignals(true);
    m_list->clear();
    m_rowToConv.clear();

    Logger::instance().debug(QString("[ConvList] refreshList: rendering %1 conversations").arg(m_conversations.size()));

    for (int i = 0; i < m_conversations.size(); ++i)
    {
        const auto &conv = m_conversations[i];

        if (!isValidConv(conv)) continue;

        m_rowToConv.append(i);
        QString title = conv.displayTitle();
        bool isDownloaded = !conv.model.isEmpty() || !conv.messages.isEmpty();

        Logger::instance().debug(QString("[ConvList] Item %1 fileId=%2 title='%3' isDownloaded=%4 messages=%5 model='%6'")
            .arg(i)
            .arg(conv.fileId)
            .arg(title)
            .arg(isDownloaded ? "true" : "false")
            .arg(conv.messages.size())
            .arg(conv.model));

        QString labelText;
        if (isDownloaded)
        {
            int tok = conv.totalTokenCount();
            QString modelShort = conv.model.section('/', -1);
            QString dateStr = conv.lastMessageTime().isValid()
                ? conv.lastMessageTime().toLocalTime().toString("yyyy-MM-dd hh:mm:ss")
                : "";

            labelText = QString("%1\n[%2] %3 tok  %4")
                            .arg(title, modelShort).arg(tok).arg(dateStr);
        }
        else
        {
            QString dateStr = conv.lastModified.isValid()
                ? conv.lastModified.toLocalTime().toString("yyyy-MM-dd hh:mm:ss")
                : "";
            labelText = QString("%1\n[Click to load...] %2").arg(title, dateStr);
        }

        Logger::instance().debug(QString("[ConvList] Label text: %1").arg(labelText));

        auto *item = new QListWidgetItem(labelText);
        item->setToolTip(conv.fileName);
        m_list->addItem(item);
    }

    m_list->blockSignals(false);
}

void ConversationListPane::applySort()
{
    switch (m_sortMode)
    {
    case NewestFirst:
        std::sort(m_conversations.begin(), m_conversations.end(),
            [](const Conversation &a, const Conversation &b)
            {
                auto ta = a.messages.isEmpty() ? a.lastModified : a.messages.last().createTime;
                auto tb = b.messages.isEmpty() ? b.lastModified : b.messages.last().createTime;
                return ta > tb;
            });
        break;
    case OldestFirst:
        std::sort(m_conversations.begin(), m_conversations.end(),
            [](const Conversation &a, const Conversation &b)
            {
                auto ta = a.messages.isEmpty() ? a.lastModified : a.messages.last().createTime;
                auto tb = b.messages.isEmpty() ? b.lastModified : b.messages.last().createTime;
                return ta < tb;
            });
        break;
    case AlphaAZ:
        std::sort(m_conversations.begin(), m_conversations.end(),
            [](const Conversation &a, const Conversation &b)
            { return a.fileName.toLower() < b.fileName.toLower(); });
        break;
    case AlphaZA:
        std::sort(m_conversations.begin(), m_conversations.end(),
            [](const Conversation &a, const Conversation &b)
            { return a.fileName.toLower() > b.fileName.toLower(); });
        break;
    }
}

void ConversationListPane::sortBy(SortMode mode)
{
    m_sortMode = mode;
    applySort();
    refreshList();
}

void FilePreviewWidget::setContent(const QByteArray &data, const QString &mimeType)
{
    m_mimeType = mimeType;

    if (mimeType.startsWith("text/") || mimeType.contains("json") || mimeType.contains("javascript"))
    {
        m_textView->setPlainText(QString::fromUtf8(data));
        m_stack->setCurrentIndex(0);
        m_stack->setVisible(true);
        m_downloadBtn->setVisible(false);
        m_viewBtn->setVisible(false);
    }
    else if (mimeType.startsWith("image/"))
    {
        QPixmap pix;
        if (pix.loadFromData(data))
        {
            m_pixmap = pix;
            m_zoomScale = 1.0;
            m_imageView->setPixmap(pix.scaledToWidth(400, Qt::SmoothTransformation));
            m_stack->setCurrentIndex(1);
            m_stack->setVisible(true);
            m_downloadBtn->setVisible(true);
            m_viewBtn->setVisible(true);
            m_imageToolbar->setVisible(true);
        }
    }
    else
    {
        m_stack->setVisible(false);
        m_downloadBtn->setVisible(true);
        m_viewBtn->setVisible(false);
    }
}

void MainWindow::onSendPrompt()
{
    QString text = m_chatInput->text().trimmed();
    if (text.isEmpty() || m_waitingForResponse)
    {
        return;
    }

    m_chatInput->clear();
    m_chatInput->setEnabled(false);
    m_sendButton->setEnabled(false);
    m_waitingForResponse = true;

    if (m_currentLocalConvIndex < 0)
    {
        Conversation conv;
        conv.fileId = QString("local-%1").arg(QDateTime::currentMSecsSinceEpoch());
        conv.fileName = text.left(80);
        conv.model = "copilot";
        m_localConversations.append(conv);
        m_currentLocalConvIndex = m_localConversations.size() - 1;
        m_convPane->addConversation(conv);
    }

    addCopilotMessage("user", text);

    if (m_currentLocalConvIndex >= 0)
    {
        auto &conv = m_localConversations[m_currentLocalConvIndex];
        m_convPane->addConversation(conv);
        m_promptPane->setConversation(conv);
    }

    statusBar()->showMessage("Sending...");
    m_copilot->sendMessage(text, m_currentConversationId);
}

void MainWindow::addCopilotMessage(const QString &role, const QString &text)
{
    if (m_currentLocalConvIndex < 0)
        return;

    auto &conv = m_localConversations[m_currentLocalConvIndex];

    Message msg;
    msg.text = text;
    msg.role = role;
    msg.createTime = QDateTime::currentDateTime();
    conv.messages.append(msg);

    if (role == "user")
    {
        Conversation::PromptResponsePair pair;
        pair.prompt = msg;
        conv.pairs.append(pair);
    }
    else if (!conv.pairs.isEmpty())
    {
        conv.pairs.last().responses.append(msg);
    }
}

void MainWindow::onSignIn()
{
    QString userProfilePath = QStandardPaths::writableLocation(QStandardPaths::DocumentsLocation);
    QString configPath = userProfilePath + "/AI/oauth_config.json";

    if (QFileInfo::exists(configPath))
    {
        m_auth->loadConfig(configPath);
    }
    else
    {
        bool ok;
        QString clientId = QInputDialog::getText(this, "Google API Credentials",
            "oauth_config.json not found at:\n" + configPath + "\n\n"
            "Enter your Google Drive Client ID:",
            QLineEdit::Normal, QString(), &ok);
        if (!ok || clientId.trimmed().isEmpty())
            return;

        QString clientSecret = QInputDialog::getText(this, "Google API Credentials",
            "Enter your Google Drive Client Secret:",
            QLineEdit::Password, QString(), &ok);
        if (!ok || clientSecret.trimmed().isEmpty())
            return;

        m_auth->setCredentials(clientId.trimmed(), clientSecret.trimmed());

        QDir().mkpath(QFileInfo(configPath).absolutePath());
        QFile f(configPath);
        if (f.open(QIODevice::WriteOnly))
        {
            QJsonObject installed;
            installed["client_id"] = clientId.trimmed();
            installed["client_secret"] = clientSecret.trimmed();
            QJsonObject root;
            root["installed"] = installed;
            f.write(QJsonDocument(root).toJson());
            f.close();
        }
    }

    if (!m_auth->isAuthenticated())
        m_auth->authenticate();
    else
    {
        m_drive->setAccessToken(m_auth->accessToken());
        m_drive->findFolder("Google AI Studio");
    }
}

void MainWindow::onSignOut()
{
    m_auth->signOut();

}

void MainWindow::onConversationsLoaded(const QList<DriveFile> &files)

void MainWindow::onConversationSelected(int index)
{
    Logger::instance().debug(QString("[MainWindow] onConversationSelected: index=%1").arg(index));

    if (index < 0 || index >= m_convPane->conversationCount())
    {
        Logger::instance().debug("[MainWindow] Index out of range, returning");
        return;
    }

    m_selectedConvIndex = index;
    const auto &conv = m_convPane->conversationAt(index);

    Logger::instance().debug(QString("[MainWindow] Selected: fileId=%1 fileName=%2 model=%3 messages=%4 pairs=%5")
        .arg(conv.fileId, conv.fileName, conv.model)
        .arg(conv.messages.size())
        .arg(conv.pairs.size()));

    if (conv.model.isEmpty() && conv.messages.isEmpty())
    {
        Logger::instance().debug("[MainWindow] Stub detected, triggering download");
        statusBar()->showMessage("Downloading conversation data...");
        m_promptPane->clear();
        m_responsePane->clear();
        m_drive->downloadConversation(conv.fileId, conv.fileName);
        return;
    }

    Logger::instance().debug("[MainWindow] Conversation fully loaded, displaying");
    m_promptPane->setConversation(conv);
    m_responsePane->clear();

    if (!conv.pairs.isEmpty())
        onPromptSelected(0);

    statusBar()->showMessage(
        QString("Conversation: %1 | %2 pairs | %3 tokens")
            .arg(conv.fileName)
            .arg(conv.pairs.size())
            .arg(conv.totalTokenCount()));
}

void MainWindow::onPromptSelected(int pairIndex)
{
    if (m_selectedConvIndex < 0)
    {
        return;
    }

    const auto &conv = m_convPane->conversationAt(m_selectedConvIndex);
    if (pairIndex < 0 || pairIndex >= conv.pairs.size())
    {
        return;
    }

    const auto &pair = conv.pairs.at(pairIndex);
    bool showThoughts = m_toggleThoughts->isChecked();

    m_responsePane->showResponse(pair, showThoughts, m_drive);
}

void MainWindow::onSearchTriggered()
{
    if (!m_searchDlg)
        m_searchDlg = new SearchDialog(m_convPane->allConversations(), this);

    m_searchDlg->show();
    m_searchDlg->raise();
    m_searchDlg->activateWindow();
}

void MainWindow::onToggleThoughts(bool visible)
{
    Q_UNUSED(visible)
    if (m_selectedConvIndex >= 0)
    {
        int promptIndex = m_promptPane->currentIndex();
        if (promptIndex >= 0)
            onPromptSelected(promptIndex);
    }
}

void MainWindow::onSortChanged(int index)
{
    switch (index)
    {
    case 0: m_convPane->sortBy(ConversationListPane::NewestFirst); break;
    case 1: m_convPane->sortBy(ConversationListPane::OldestFirst); break;
    case 2: m_convPane->sortBy(ConversationListPane::AlphaAZ); break;
    case 3: m_convPane->sortBy(ConversationListPane::AlphaZA); break;
    }
}

void MainWindow::loadConversation(const QString &fileId, const QString &fileName)
{
    m_drive->downloadConversation(fileId, fileName);
}

    m_list->clear();
    m_currentConv = Conversation();

void PromptListPane::refreshList()
{
    m_list->blockSignals(true);
    m_list->clear();

    for (int i = 0; i < m_currentConv.pairs.size(); ++i)
    {
        const auto &pair = m_currentConv.pairs[i];
        QString preview = pair.prompt.text.left(80);
        if (pair.prompt.text.length() > 80)
            preview += "...";
        if (preview.isEmpty() && !pair.prompt.driveDocumentId.isEmpty())
            preview = "[Uploaded file]";
        else if (preview.isEmpty() && !pair.prompt.inlineImageData.isEmpty())
            preview = "[Inline image]";
        else if (preview.isEmpty())
            preview = "(empty prompt)";

        QString dateStr = pair.prompt.createTime.isValid()
            ? pair.prompt.createTime.toLocalTime().toString("yyyy-MM-dd hh:mm:ss")
            : "no-time";

        auto *item = new QListWidgetItem(
            QString("%1\n[%2 tok]  %3")
                .arg(preview)
                .arg(pair.prompt.tokenCount)
                .arg(dateStr));
        m_list->addItem(item);

        if (!pair.prompt.driveDocumentId.isEmpty())
            Logger::instance().info(
                QString("[Attachment] PromptList: pair %1 docId='%2' text='%3'")
                    .arg(i)
                    .arg(pair.prompt.driveDocumentId)
                    .arg(pair.prompt.text.left(60)));
    }

    m_list->blockSignals(false);
}

// Convert Google AI markdown to clean HTML for QTextBrowser
static QString markdownToHtml(const QString &markdown)
{
    QString md = markdown;

    // --- Pre-process features Qt's markdown doesn't handle ---

    // Strikethrough: ~~text~~ → <s>text</s>
    md.replace(QRegularExpression("~~(.+?)~~"), "<s>\\1</s>");

    // Task list items: use Unicode ballot boxes
    md.replace("- [x]", "- &#x2611;");
    md.replace("- [ ]", "- &#x2610;");

    // Convert via Qt's markdown renderer
    QTextDocument doc;
    doc.setMarkdown(md);
    QString html = doc.toHtml();

    // Extract only <body> content, strip the outer html/head/body tags
    int bodyStart = html.indexOf("<body");
    if (bodyStart < 0)
        return html;
    bodyStart = html.indexOf(">", bodyStart) + 1;
    int bodyEnd = html.indexOf("</body>", bodyStart);
    if (bodyEnd < 0)
        return html;
    QString body = html.mid(bodyStart, bodyEnd - bodyStart);

    // Clean up Qt-specific cruft: remove excessive <span> font tags
    body.replace(QRegularExpression("<span[^>]*>"), "");
    body.replace("</span>", "");
    // QTextDocument adds <p style="..."> — keep the style but clean up orphans
    body.replace("<p style=\"-qt-paragraph-type:empty\"/>", "<p><br></p>");

    return body;
}

ResponsePane::ResponsePane(QWidget *parent)
    : QWidget(parent)
    , m_browser(new QTextBrowser(this))
{
    m_layout = new QVBoxLayout(this);
    m_layout->setContentsMargins(0, 0, 0, 0);

void ResponsePane::showResponse(const Conversation::PromptResponsePair &pair,
                                bool showThoughts, DriveApi *drive)
{
    removeFilePreviews();
    m_browser->clear();

    m_browser->setHtml(
        "<h3>Prompt</h3><div style='padding:8px; margin:4px 0;'>"
        + markdownToHtml(pair.prompt.text) + "</div><hr>");

    if (!pair.prompt.driveDocumentId.isEmpty())
    {
        Logger::instance().info(
            QString("[Attachment] showResponse: displaying pair with docId='%1'")
                .arg(pair.prompt.driveDocumentId));
        appendFileAttachment(pair.prompt.driveDocumentId, drive);
    }

    if (!pair.prompt.inlineImageData.isEmpty())
    {
        appendInlineImage(pair.prompt.inlineImageData, pair.prompt.inlineImageMime);
    }

    for (const auto &resp : pair.responses)
    {
        appendModelResponse(resp.text, resp.isThought, showThoughts);
    }

    if (pair.responses.isEmpty())
        m_browser->append("<p><i>No response yet.</i></p>");

    m_browser->verticalScrollBar()->setValue(0);
}

void ResponsePane::appendModelResponse(const QString &text, bool isThought, bool showThoughts)
{
    if (isThought && !showThoughts)
        return;

    QString html;

    if (text.startsWith("ERROR: Cannot read"))
    {
        html = "<div style='background:#fff3cd; border:1px solid #ffc107; border-radius:4px; "
               "padding:8px; margin:4px 0;'>"
               "<b>&#x26A0; Image not supported:</b><br>" + markdownToHtml(text) + "</div>";
    }
    else if (isThought)
    {
        html = "<div style='background:#f0f0f0; border-left:3px solid #888; padding:8px; margin:4px 0;'>"
               "<i>Thought:</i><br>" + markdownToHtml(text) + "</div>";
    }
    else
    {
        html = "<div style='padding:8px; margin:4px 0;'>" + markdownToHtml(text) + "</div>";
    }

    m_browser->append(html);
}

void ResponsePane::appendFileAttachment(const QString &fileId, DriveApi *drive)
{
    if (!drive)
    {
        Logger::instance().warn(
            QString("[Attachment] appendFileAttachment: drive is null for fileId='%1'").arg(fileId));
        return;
    }

    Logger::instance().info(
        QString("[Attachment] appendFileAttachment: downloading fileId='%1'").arg(fileId));
    m_browser->append("<hr><b>Attached file:</b> " + fileId + "<br>");

    auto *preview = new FilePreviewWidget(fileId, this);
    m_layout->addWidget(preview);
    m_filePreviews.append(preview);

    QPointer<FilePreviewWidget> safePreview(preview);

    connect(drive, &DriveApi::attachmentDownloaded, this,
        [this, safePreview, fileId](const QString &id, const QByteArray &data, const QString &mimeType)
        {
            if (id == fileId && safePreview)
                safePreview->setContent(data, mimeType);
        });

    drive->downloadAttachment(fileId);
}

void ResponsePane::appendInlineImage(const QString &base64data, const QString &mimeType)
{
    QByteArray raw = QByteArray::fromBase64(base64data.toUtf8());
    QPixmap pix;
    if (pix.loadFromData(raw))
    {
        auto *preview = new FilePreviewWidget("[inline image]", this);
        preview->setContent(raw, mimeType);
        m_layout->addWidget(preview);
        m_filePreviews.append(preview);
    }
    else
    {
        m_browser->append("<p><i>[Inline image could not be decoded]</i></p>");
    }
}

void ResponsePane::removeFilePreviews()
{
    for (auto *w : m_filePreviews)
    {
        m_layout->removeWidget(w);
        w->deleteLater();
    }
    m_filePreviews.clear();
}

void ResponsePane::clear()
{
    removeFilePreviews();
    m_browser->clear();
}

SearchDialog::SearchDialog(const QList<Conversation> &conversations,
                           QWidget *parent)
    : QDialog(parent)
    , m_searchInput(new QLineEdit(this))
    , m_resultsList(new QListWidget(this))
    , m_conversations(conversations)
{
    setWindowTitle("Search Conversations");
    resize(500, 400);

    auto *layout = new QVBoxLayout(this);
    layout->addWidget(new QLabel("Search across all conversations:", this));
    layout->addWidget(m_searchInput);
    layout->addWidget(m_resultsList);

    m_searchInput->setPlaceholderText("Type to search...");
    connect(m_searchInput, &QLineEdit::textChanged, this, &SearchDialog::onSearchTextChanged);
}

void SearchDialog::onSearchTextChanged(const QString &text)
{
    m_resultsList->clear();

    if (text.length() < 2)
        return;

    QString lower = text.toLower();

    for (const auto &conv : m_conversations)
    {
        for (const auto &msg : conv.messages)
        {
            if (msg.text.toLower().contains(lower))
            {
                QString preview = msg.text.left(120);
                if (msg.text.length() > 120)
                    preview += "...";

                auto *item = new QListWidgetItem(
                    QString("[%1] %2: %3")
                        .arg(conv.fileName, msg.role, preview));
                m_resultsList->addItem(item);
            }
        }
    }

    if (m_resultsList->count() == 0 && text.length() >= 2)
        m_resultsList->addItem("No matches found.");
}
